# CIRCUIT: bud-place-only bud detects place, latency 5 gt, pulse 2 gt, re-arm 6 gt, footprint 1 cell, tileable every 2 (0 0 2), 4 blocks, 2x4x1, robust
name: bud-place-only
circuit: bud
tile: 0 0 2
# Tells when a block is placed at the target, and ignores its removal and the target's own
# changes: 4 blocks. A redstone torch under the target powers any solid block placed there, which
# powers the sticky piston beside it; the piston lifts its observer, which fires once as it lands,
# into the output cell behind it (a 2-tick pulse). On removal the piston pulls the observer back
# down and it fires one cell lower, away from the output. Detail: turning the block in place, a
# shape-only change or a pure update leaves it powered: not detected.
origin: 0 -1 0
port target: 0 0 0
port output: 0 2 0
palette:
  T minecraft:redstone_wall_torch[facing=west,lit=true]
  B minecraft:stone
  P minecraft:sticky_piston[extended=false,facing=up]
  O minecraft:observer[facing=east,powered=false]
layer 0:
  TB
layer 1:
  .P
layer 2:
  .O
