name: pulse-dual-sculk-cal-L10
circuit: dual-edge
# One 10-tick pulse (half a second) every time the input turns on AND every time it turns off,
# from just 2 blocks, for quiet builds: the input opens and closes a trapdoor, and a sound sensor
# (calibrated sculk sensor) switches on for 10 ticks at each sound. It rests 10 ticks after each
# pulse, so the input must stay on or off at least 30 ticks, and it hears footsteps and dropped
# items within 16 blocks too. The smallest 10 gt dual-edge detector.
# Detail: block_open (10) and block_close (9); nothing at the sensor's back: every frequency.
origin: 0 0 0
port input: 0 0 0
port output: 1 0 1
palette:
  T minecraft:oak_trapdoor[facing=north,half=bottom,open=false,powered=false,waterlogged=false]
  K minecraft:calibrated_sculk_sensor[facing=west,power=0,sculk_sensor_phase=inactive,waterlogged=false]
layer 0:
  .T.
  ..K
