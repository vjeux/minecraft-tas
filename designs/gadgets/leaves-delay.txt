name: gadget-leaves-delay
# A 1 gt delay element: persistent leaves F recompute their `distance` in a scheduled tick 1 gt
# after a neighbor change (LeavesBlock.updateShape -> scheduleTick(pos, this, 1)). Lever L fires
# piston P in tick 1, log W leaves F's side (F was distance 1); F turns distance 7 in tick 2 and
# observer O watching F fires ticks 4..6. Every redstone component delays by an even number of
# game ticks (2, 4, 6, 8); this 1 gt element (or anything else scheduling 1 gt: scaffolding,
# 26.3's sugar cane/cactus/bamboo survival checks) shifts a chain by one tick. persistent=true
# leaves never decay (no random ticks).
origin: -2 0 0
palette:
  S minecraft:stone
  L minecraft:lever[face=floor,facing=east,powered=false]
  P minecraft:piston[extended=false,facing=east]
  W minecraft:oak_log[axis=y]
  F minecraft:oak_leaves[distance=1,persistent=true,waterlogged=false]
  O minecraft:observer[facing=north,powered=false]
layer 0:
  S
layer 1:
  LPW
  ..F
  ..O
